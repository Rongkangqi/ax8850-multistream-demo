cmake_minimum_required(VERSION 3.18)
project(ax8850_multistream_demo VERSION 1.0.0 LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Build on a Linux AXCL host. Windows is only used to store/edit this repository.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(AXCL_ROOT /usr CACHE PATH "Installed AXCL SDK root")
find_path(AXCL_INCLUDE axcl.h PATHS "${AXCL_ROOT}/include/axcl" "${AXCL_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
find_library(AXCL_RUNTIME axcl_rt PATHS "${AXCL_ROOT}/lib/axcl" "${AXCL_ROOT}/lib" "${AXCL_ROOT}/lib64" NO_DEFAULT_PATH REQUIRED)
get_filename_component(AXCL_LIBRARY_DIR "${AXCL_RUNTIME}" DIRECTORY)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc)
find_package(Threads REQUIRED)
set(PIPELINE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/ax-pipeline")
set(NPU "${PIPELINE}/plugins/common/npu")
set(AXSDK_CHIP_TYPE axcl CACHE STRING "" FORCE)
set(AXSDK_AXCL_DIR "${AXCL_ROOT}" CACHE PATH "" FORCE)
set(AXSDK_BUILD_SHARED ON CACHE BOOL "" FORCE)
set(AXSDK_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(AXSDK_BUILD_SMOKE_TESTS OFF CACHE BOOL "" FORCE)
set(AXSDK_BUILD_CAPI OFF CACHE BOOL "" FORCE)
set(BUILD_APPS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/ax-pipeline/deps/ax-video-sdk sdk)
add_subdirectory(vendor/ax-pipeline/third-party/ByteTrack bytetrack)

add_library(demo_tracking STATIC "${PIPELINE}/src/tracking/ax_bytetrack.cpp")
target_include_directories(demo_tracking PUBLIC "${PIPELINE}/include" PRIVATE "${PIPELINE}/third-party/ByteTrack/include")
target_link_libraries(demo_tracking PUBLIC ByteTrack)
add_library(demo_npu STATIC
  "${NPU}/src/npu/models/ax_model_base.cpp"
  "${NPU}/src/npu/models/ax_model_det.cpp"
  "${NPU}/src/npu/runner/logger.cpp"
  "${NPU}/src/npu/runner/axcl/axcl_manager.cpp"
  patches/npu/ax_model_runner_axcl.cpp)
target_include_directories(demo_npu PUBLIC "${NPU}/include" "${CMAKE_CURRENT_SOURCE_DIR}/patches/npu"
  "${NPU}/src/npu/runner/axcl" PRIVATE "${NPU}/src/npu/runner")
target_compile_definitions(demo_npu PUBLIC AXPIPELINE_HAVE_AXCL=1 AXPIPELINE_DEFAULT_BACKEND_AXCL=1)
target_link_libraries(demo_npu PUBLIC ax_video_sdk ax_video_sdk::ax_msp axcl_rt axcl_npu)
target_link_options(demo_npu PUBLIC -Wl,--allow-shlib-undefined)

foreach(plugin pcd yolo26 yolov8_split)
  if(plugin STREQUAL pcd)
    set(plugin_dir pcd.axera)
  else()
    set(plugin_dir "${plugin}")
  endif()
  add_library(ax_plugin_${plugin} SHARED "${PIPELINE}/plugins/${plugin_dir}/plugin_${plugin}.cpp")
  target_include_directories(ax_plugin_${plugin} PRIVATE "${PIPELINE}/include" "${PIPELINE}/third-party/json")
  target_compile_definitions(ax_plugin_${plugin} PRIVATE AX_PLUGIN_BUILD_DLL=1)
  target_link_libraries(ax_plugin_${plugin} PRIVATE demo_npu demo_tracking)
  set_target_properties(ax_plugin_${plugin} PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib/plugins"
    INSTALL_RPATH "$ORIGIN/..;${AXCL_LIBRARY_DIR}")
  install(TARGETS ax_plugin_${plugin} LIBRARY DESTINATION lib/plugins COMPONENT Demo)
endforeach()
add_executable(six_app src/six_app.cpp)
target_include_directories(six_app PRIVATE "${PIPELINE}/include" "${PIPELINE}/third-party/json"
  "${PIPELINE}/deps/ax-video-sdk/src/common" ${OpenCV_INCLUDE_DIRS})
target_compile_options(six_app PRIVATE -Wall -Wextra)
target_link_libraries(six_app PRIVATE demo_npu demo_tracking axcl_ivps ${OpenCV_LIBS} Threads::Threads ${CMAKE_DL_LIBS})
set_target_properties(six_app PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
  INSTALL_RPATH "$ORIGIN/../lib;${AXCL_LIBRARY_DIR}")
set_target_properties(ax_video_sdk PROPERTIES INSTALL_RPATH "$ORIGIN;${AXCL_LIBRARY_DIR}")
install(TARGETS six_app RUNTIME DESTINATION bin COMPONENT Demo)
install(TARGETS ax_video_sdk LIBRARY DESTINATION lib COMPONENT Demo)
